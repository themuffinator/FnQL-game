# G3: --cheats --factory tdm --set g_runes 1 --frames 2200.
# All four rune states, one-rune ownership, Scout/Haste priority and armor
# decay, Guard userinfo, Armor Regen without ammo regeneration, manual and
# death drops, cross-player pickup, ten/30-second relocation, map restart.

at 0 bravo weapon=2

at 0 delta weapon=2

clientcmd 16 bravo setviewpos -700 320 40 0
clientcmd 17 charlie setviewpos -400 -320 40 270
clientcmd 18 delta setviewpos 120 -320 40 0

at 400 charlie buttons=0

clientcmd 200 alpha droprune
clientcmd 202 alpha setviewpos -400 320 40 180
clientcmd 220 bravo droprune

clientcmd 280 bravo droprune

# Death must toss a rune that can relocate after thirty seconds.
clientcmd 450 delta kill

at 494 delta buttons=0

clientcmd 700 charlie droprune

clientcmd 1700 alpha score
